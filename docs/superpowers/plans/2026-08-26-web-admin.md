# Web Admin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Небольшой веб-сервис рядом с ботом: статус подключения, выбор и редактирование персон, просмотр собранного системного промпта, списки доступа, рантайм-настройки, пауза, лента последних ответов с удалением, песочница без отправки в VK.

**Architecture:** Бот и веб — два процесса в одном docker-compose, общий язык — файлы. Бот читает `config/runtime.json`, `config/personas/*.md`, `config/access.json` по mtime (как уже делает `AccessFile`) и пишет `state/status.json` (heartbeat) и `state/replies.jsonl` (лента). Веб редактирует `config/`, читает `state/`, и только в песочнице/удалении ходит в LLM/VK напрямую. Вся логика — чистые функции в `bot/` и `web/services.py`, HTTP-слой тонкий.

**Tech Stack:** Python 3.12, FastAPI + Jinja2 + htmx (без сборки фронта), uvicorn, pytest + `fastapi.testclient` (httpx). Существующие `vk_api`, `openai`, `pydantic-settings`.

**Spec:** раздел «Design» ниже (согласованные в чате рекомендации 1–10).

## Global Constraints

- Порядок слоёв системного промпта неизменен: `SYSTEM_PROMPT > user_prompts > private_prompt > persona`. Меняется только источник слоя persona (файл вместо константы).
- Тексты чужих сообщений не пишутся ни в логи, ни в `state/`, если `LOG_MESSAGE_TEXT` не включён. Свой ответ бота в ленте хранить можно.
- Веб слушает только `127.0.0.1` (compose `ports: "127.0.0.1:8080:8080"`) и требует Basic Auth с паролем из env `WEB_PASSWORD`. Без пароля сервис не стартует.
- Все правки конфигов из веба идут через файлы в `config/` атомарно (`tmp` + `os.replace`), чтобы hot-reload бота оставался единственным механизмом применения.
- Комментарии в коде — только «почему» (правило 4 CLAUDE.md). Коммиты — одна строка на английском, без LLM-атрибуции (правило 5).
- Лимиты `RateLimiter` остаются в `.env` (нужен пересбор объекта); в рантайме управляются: `active_persona`, `paused`, `paused_peers`, `temperature`, `reply_delay_min/max`, `context_messages`.

### Файлы данных

`config/runtime.json` (пишет веб, читает бот; нет файла — дефолты из `Settings`):
```json

  "paused": false,
```

```json
 "started_at": 1756200000.0, "last_event_at": 1756203000.0, "last_reply_at": 1756202900.0,
```

`state/replies.jsonl` — по строке на ответ: `{"ts", "peer_id", "from_id", "message_id", "reply_message_id", "reply", "tag": null|str, "latency_s", "persona"}`; `tag` заполняется только при `LOG_MESSAGE_TEXT`. Файл усекается до последних 200 строк при записи.

```python
# bot/hotfile.py
import logging
import os
from typing import Callable, Generic, TypeVar

log = logging.getLogger("vk-bot")

T = TypeVar("T")


class WatchedFile(Generic[T]):
    """Значение из файла, перечитываемое по mtime без перезапуска.
    Нет файла — default; битый файл — прежнее значение до следующей записи."""

    def __init__(self, path: str, parse: Callable[[str], T], default: T, log_label: str):
        self._path = path
        self._parse = parse
        self._default = default
        self._label = log_label
        self._value = default
        self._mtime: float | None = None

    def current(self) -> T:
        try:
            mtime = os.stat(self._path).st_mtime
        except OSError:
            if self._mtime is not None:
                log.info("Файл %s пропал, работаю без него", self._path)
            self._mtime = None
            self._value = self._default
            return self._value

        if mtime != self._mtime:
            # mtime запоминаем и при ошибке, чтобы не перечитывать битый файл на каждом событии
            self._mtime = mtime
            try:
                with open(self._path, encoding="utf-8") as f:
                    self._value = self._parse(f.read())
                log.info("%s перечитан из %s", self._label, self._path)
            except Exception:
                log.exception("Не удалось прочитать %s, работаю по прежнему значению", self._path)
        return self._value
```

```python
# bot/runtime.py
import json
from dataclasses import asdict, dataclass

from bot.hotfile import WatchedFile


@dataclass(frozen=True)
class RuntimeSettings:
    active_persona: str = ""
    paused: bool = False
    paused_peers: frozenset[int] = frozenset()
    temperature: float = 0.8
    reply_delay_min: float = 2
    reply_delay_max: float = 4
    context_messages: int = 60


def runtime_defaults(settings) -> RuntimeSettings:
    return RuntimeSettings(
        temperature=settings.llm_temperature,
        reply_delay_min=settings.reply_delay_min,
        reply_delay_max=settings.reply_delay_max,
        context_messages=settings.context_messages,
    )


def parse_runtime(data: dict, defaults: RuntimeSettings) -> RuntimeSettings:
    def get(key, cast):
        value = data.get(key)
        return getattr(defaults, key) if value is None else cast(value)

    lo = get("reply_delay_min", float)
    hi = get("reply_delay_max", float)
    return RuntimeSettings(
        active_persona=get("active_persona", str).strip(),
        paused=get("paused", bool),
        paused_peers=frozenset(int(p) for p in data.get("paused_peers") or []) or defaults.paused_peers,
        temperature=get("temperature", float),
        reply_delay_min=min(lo, hi),
        reply_delay_max=max(lo, hi),
        context_messages=get("context_messages", int),
    )


def dump_runtime(rt: RuntimeSettings) -> dict:
    data = asdict(rt)
    data["paused_peers"] = sorted(rt.paused_peers)
    return data


class RuntimeFile:
    def __init__(self, path: str, defaults: RuntimeSettings):
        self._file = WatchedFile(
            path,
            parse=lambda text: parse_runtime(json.loads(text), defaults),
            default=defaults,
            log_label="Рантайм-настройки",
        )

    def current(self) -> RuntimeSettings:
        return self._file.current()
```

```python
# bot/personas.py
import os
import re
from dataclasses import dataclass

from bot.hotfile import WatchedFile

SLUG_RE = re.compile(r"^[a-z0-9_-]{1,40}$")


@dataclass(frozen=True)
class Persona:
    slug: str
    name: str
    prompt: str


def parse_persona(slug: str, text: str) -> Persona:
    lines = text.strip().splitlines()
    if lines and lines[0].startswith("# "):
        return Persona(slug, lines[0][2:].strip(), "\n".join(lines[1:]).strip())
    return Persona(slug, slug, text.strip())


def render_persona(name: str, prompt: str) -> str:
    return f"# {name.strip()}\n\n{prompt.strip()}\n"


def persona_path(personas_dir: str, slug: str) -> str:
    if not SLUG_RE.match(slug):
        raise ValueError(f"Недопустимый slug персоны: {slug!r}")
    return os.path.join(personas_dir, f"{slug}.md")


def list_personas(personas_dir: str) -> list[Persona]:
    try:
        names = sorted(os.listdir(personas_dir))
    except OSError:
        return []
    personas = []
    for name in names:
        slug, ext = os.path.splitext(name)
        if ext == ".md" and SLUG_RE.match(slug):
            with open(os.path.join(personas_dir, name), encoding="utf-8") as f:
                personas.append(parse_persona(slug, f.read()))
    return personas


def load_persona(personas_dir: str, slug: str) -> Persona | None:
    try:
        with open(persona_path(personas_dir, slug), encoding="utf-8") as f:
            return parse_persona(slug, f.read())
    except (OSError, ValueError):
        return None


class PersonaFile:
    def __init__(self, personas_dir: str, slug: str, fallback_prompt: str):
        path = persona_path(personas_dir, slug) if SLUG_RE.match(slug) else os.path.join(personas_dir, "_.md")
        self._file = WatchedFile(
            path,
            parse=lambda text: parse_persona(slug, text).prompt,
            default=fallback_prompt,
            log_label=f"Персона {slug}",
        )

    def current(self) -> str:
        return self._file.current()
```

```python
# bot/status.py
import json
import os
import time
from collections import deque
from dataclasses import asdict, dataclass

FLUSH_INTERVAL = 5.0
ERROR_WINDOW = 3600.0


def write_json_atomic(path: str, data: dict) -> None:
    os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
    tmp = f"{path}.tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False)
    os.replace(tmp, path)


def read_json(path: str, default):
    try:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    except (OSError, ValueError):
        return default


class StatusWriter:
    def __init__(self, path: str, me_id: int, me_name: str, provider: str, model: str, now=time.time):
        self._path = path
        self._now = now
        self._errors: deque[float] = deque()
        self._last_flush = 0.0
        self._data = {
            "me_id": me_id, "me_name": me_name, "provider": provider, "model": model,
            "started_at": now(), "last_event_at": None, "last_reply_at": None,
            "errors_last_hour": 0, "active_persona": "", "paused": False,
        }

    def event(self) -> None:
        self._data["last_event_at"] = self._now()

    def replied(self) -> None:
        self._data["last_reply_at"] = self._now()

    def error(self) -> None:
        self._errors.append(self._now())

    def set_runtime(self, active_persona: str, paused: bool) -> None:
        self._data["active_persona"] = active_persona
        self._data["paused"] = paused

    def flush(self, force: bool = False) -> None:
        now = self._now()
        if not force and now - self._last_flush < FLUSH_INTERVAL:
            return
        while self._errors and now - self._errors[0] > ERROR_WINDOW:
            self._errors.popleft()
        self._data["errors_last_hour"] = len(self._errors)
        write_json_atomic(self._path, self._data)
        self._last_flush = now


@dataclass(frozen=True)
class ReplyRecord:
    ts: float
    peer_id: int
    from_id: int
    message_id: int
    reply_message_id: int
    reply: str
    tag: str | None
    latency_s: float
    persona: str


class ReplyLog:
    def __init__(self, path: str, keep: int = 200):
        self._path = path
        self._keep = keep

    def append(self, rec: ReplyRecord) -> None:
        lines = self._lines()
        lines.append(json.dumps(asdict(rec), ensure_ascii=False))
        os.makedirs(os.path.dirname(self._path) or ".", exist_ok=True)
        tmp = f"{self._path}.tmp"
        with open(tmp, "w", encoding="utf-8") as f:
            f.write("\n".join(lines[-self._keep:]) + "\n")
        os.replace(tmp, self._path)

    def read(self) -> list[ReplyRecord]:
        records = []
        for line in self._lines():
            try:
                records.append(ReplyRecord(**json.loads(line)))
            except (ValueError, TypeError):
                continue
        return list(reversed(records))

    def _lines(self) -> list[str]:
        try:
            with open(self._path, encoding="utf-8") as f:
                return [ln for ln in f.read().splitlines() if ln.strip()]
        except OSError:
            return []
```

```python
def sender_id(event, me_id: int) -> int | None:
    # в личном диалоге long poll кладёт в user_id собеседника даже для своих сообщений
    if getattr(event, "from_me", False):
        return me_id
    return getattr(event, "user_id", None)


def persona_prompt_for(deps: BotDeps, slug: str) -> str:
    if not slug:
        return deps.settings.persona_prompt
    if slug not in deps.personas:
        deps.personas[slug] = PersonaFile(deps.settings.personas_dir, slug, deps.settings.persona_prompt)
    return deps.personas[slug].current()


def handle(event, deps: BotDeps) -> None:
    vk, settings = deps.vk, deps.settings
    text = getattr(event, "text", "") or ""
    from_id = sender_id(event, vk.me_id)
    if from_id is None:
        return
    lists = deps.access.current()
    rt = deps.runtime.current()
    deps.status.set_runtime(rt.active_persona, rt.paused)
    if rt.paused or event.peer_id in rt.paused_peers:
        return
    if not should_reply(text, event.peer_id, from_id, vk.me_id,
                        lists.blocked_users, lists.blocked_peers,
                        lists.allowed_peers, lists.auto_reply_users):

    if not deps.limiter.allow(from_id, event.peer_id):
        log.info("Rate limit: %s", describe_message(event.peer_id, from_id, text, False))
        return

    started = time.time()
    try:
        vk.set_typing(event.peer_id)
    except Exception:
        log.warning("Не удалось включить индикатор набора", exc_info=True)
    history = vk.history(event.peer_id, rt.context_messages)
    conversation = build_context(history, vk.resolve_name,
                                 settings.message_char_limit, settings.context_char_budget,
                                 vk.me_id)
    is_private = event.peer_id < GROUP_CHAT_PEER_BASE

    private_prompt, user_prompt = private_prompts(lists, event.peer_id) if is_private else ("", "")
    reply = deps.llm.reply(conversation, vk.resolve_name(from_id), is_private=is_private,
                           private_prompt=private_prompt, user_prompt=user_prompt,
                           persona_prompt=persona_prompt_for(deps, rt.active_persona),
                           temperature=rt.temperature)
    if not reply:
        log.warning("LLM вернула пустой ответ: %s",
                    describe_message(event.peer_id, from_id, text, False))
        return
    reply = strip_self_mention(sanitize_reply(reply, [settings.vk_token, settings.llm_api_key]),
                               vk.me_id)

    time.sleep(random.uniform(rt.reply_delay_min, rt.reply_delay_max))
    sent_id = vk.send_reply(event.peer_id, event.message_id, reply)
    deps.status.replied()

        llm=LlmClient(settings),
        limiter=RateLimiter(settings.user_cooldown_seconds,
                            settings.chat_limit_per_hour,
                            settings.global_limit_per_minute),
        settings=settings,
        access=AccessFile(settings.access_file, vk.resolve_screen_names),
        runtime=RuntimeFile(settings.runtime_file, runtime_defaults(settings)),
        status=StatusWriter(f"{settings.state_dir}/status.json", vk.me_id,

            for event in vk.listen():
```

```python
from fastapi import Depends, HTTPException, status

_basic = HTTPBasic()


def make_guard(password: str):
    def guard(creds: HTTPBasicCredentials = Depends(_basic)) -> None:
        if not secrets.compare_digest(creds.password.encode(), password.encode()):
            raise HTTPException(status.HTTP_401_UNAUTHORIZED, headers={"WWW-Authenticate": "Basic"})
    return guard
```

```python
from bot.status import write_json_atomic


def load_runtime_dict(path: str) -> dict:
    return read_json(path, {})


def save_runtime_patch(path: str, **fields) -> None:
    data = load_runtime_dict(path)
    data.update(fields)
    write_json_atomic(path, data)


def save_persona(personas_dir: str, slug: str, name: str, prompt: str) -> None:
    path = persona_path(personas_dir, slug)

    os.replace(tmp, path)
```

```python
from bot.access import join_lines
from bot.llm import PERSONA_LAYER, PRIVATE_LAYER, SYSTEM_LAYER, USER_LAYER, compose_system_message
from bot.personas import load_persona


@dataclass(frozen=True)
class PromptView:
    persona_slug: str
    persona_name: str
    user_key: str | None
    message: str
    layers: list[tuple[str, str]]


def access_raw(path: str) -> dict:
    return read_json(path, {})


def prompt_view(settings, user_key: str | None) -> PromptView:
    slug = load_runtime_dict(settings.runtime_file).get("active_persona", "")
    persona = load_persona(settings.personas_dir, slug) if slug else None
    persona_name = persona.name if persona else "по умолчанию"
    persona_prompt = persona.prompt if persona else settings.persona_prompt
    raw = access_raw(settings.access_file)
    user_prompt = join_lines((raw.get("user_prompts") or {}).get(user_key)) if user_key else ""
    private_prompt = join_lines(raw.get("private_prompt")) if user_key else ""
    layers = [(SYSTEM_LAYER, settings.system_prompt), (USER_LAYER, user_prompt),
              (PRIVATE_LAYER, private_prompt), (PERSONA_LAYER, persona_prompt)]
    return PromptView(
        persona_slug=slug, persona_name=persona_name, user_key=user_key,
        message=compose_system_message(settings.system_prompt, user_prompt, private_prompt, persona_prompt),
        layers=[(t, x.strip()) for t, x in layers if x.strip()],
    )
```

```python
from bot.access import parse_access

ACCESS_LIST_FIELDS = ("blocked_users", "blocked_peers", "allowed_peers", "auto_reply_users")


def _lines(text: str) -> list[str]:
    return [ln.strip() for ln in (text or "").splitlines() if ln.strip()]


def parse_access_form(form: dict) -> dict:
    user_prompts = {}
    for line in _lines(form.get("user_prompts", "")):
        key, sep, value = line.partition(":")
        if not sep or not key.strip():
            raise ValueError(f"Строка user_prompts без «ник: текст»: {line!r}")
        user_prompts[key.strip()] = value.strip()
    data = {field: _lines(form.get(field, "")) for field in ACCESS_LIST_FIELDS}
    data["private_prompt"] = _lines(form.get("private_prompt", ""))
    data["user_prompts"] = user_prompts
    parse_access(data, resolve_screen_names=lambda names: frozenset())
    return data


def access_form_values(path: str) -> dict:
    raw = access_raw(path)
    values = {field: "\n".join(str(x) for x in raw.get(field) or []) for field in ACCESS_LIST_FIELDS}
    values["private_prompt"] = join_lines(raw.get("private_prompt"))
    values["user_prompts"] = "\n".join(f"{k}: {join_lines(v)}" for k, v in (raw.get("user_prompts") or {}).items())
    return values
```

```python
from bot.access import AccessFile, private_prompts
from bot.context import build_context
from bot.main import GROUP_CHAT_PEER_BASE


def sandbox_reply(settings, llm, vk, *, persona_slug: str, conversation: str, peer_id: int | None,
                  count: int, asker: str, temperature: float) -> tuple[str, str]:
    persona = load_persona(settings.personas_dir, persona_slug) if persona_slug else None
    persona_prompt = persona.prompt if persona else settings.persona_prompt
    is_private, private_prompt, user_prompt = False, "", ""
    if peer_id is not None:
        history = vk.history(peer_id, count)
        conversation = build_context(history, vk.resolve_name, settings.message_char_limit,
                                     settings.context_char_budget, vk.me_id)
        is_private = peer_id < GROUP_CHAT_PEER_BASE
        if is_private:
            lists = AccessFile(settings.access_file, vk.resolve_screen_names).current()
            private_prompt, user_prompt = private_prompts(lists, peer_id)
    reply = llm.reply(conversation, asker, is_private=is_private, private_prompt=private_prompt,
                      user_prompt=user_prompt, persona_prompt=persona_prompt, temperature=temperature)
    return conversation, reply
```
